# boot.txt — the scene the machine starts with: Duppo, its discs, shades and
# hues boxos_mascot.h's. Gangway shows the first frame — Duppo alone — while
# the machine starts; the display plays on from that frame: the three little
# bubbles come out of Duppo (enter) and go round it while the machine gets
# ready, the eye blinking once a round (wait), and the screen goes dark into
# the console (leave). Packed by tools/scene.py into boot.scene.

scene units=1000 paper=#000000 enter=1000 wait=4800 leave=480

paint body   radial shade=BoxDuppoBodyShade     stops=BoxDuppoBodyStops
paint socket radial shade=BoxDuppoSocketShade   stops=BoxDuppoSocketStops
paint white  radial shade=BoxDuppoWhiteShade    stops=BoxDuppoWhiteStops
paint pupil  radial shade=BoxDuppoPupilShade    stops=BoxDuppoPupilStops
paint medium radial shade=BoxDuppoMoonShades[0] stops=BoxDuppoMoonStops
paint small  radial shade=BoxDuppoMoonShades[1] stops=BoxDuppoMoonStops
paint tiny   radial shade=BoxDuppoMoonShades[2] stops=BoxDuppoMoonStops
paint dark   solid #000000

# Duppo in the logo's own units: its body in the middle of the screen, three
# tenths of the shorter side across.
group   duppo  pivot=BoxDuppoBody at=0,0 scale=8.75
ellipse body   in=duppo disc=BoxDuppoBody paint=body

# The bubbles go round the body, before it and behind the eye; each turns
# back about its own middle as far as the round turns it on, so the light
# stays where it falls on Duppo.
group   moons  in=duppo pivot=BoxDuppoBody scale=0 opacity=0
ellipse medium in=moons disc=BoxDuppoMoons[0] pivot=BoxDuppoMoons[0] paint=medium
ellipse small  in=moons disc=BoxDuppoMoons[1] pivot=BoxDuppoMoons[1] paint=small
ellipse tiny   in=moons disc=BoxDuppoMoons[2] pivot=BoxDuppoMoons[2] paint=tiny

# The eye shuts about its own middle.
group   eye    in=duppo pivot=BoxDuppoSocket
ellipse socket in=eye disc=BoxDuppoSocket paint=socket
ellipse white  in=eye disc=BoxDuppoWhite  paint=white
ellipse pupil  in=eye oval=BoxDuppoPupil  paint=pupil

# Over everything, the dark the scene leaves into.
rect    veil   box=-4000,-4000,8000,8000 paint=dark opacity=0

move moons enter scale   0:0 1000:1:out
move moons enter opacity 0:0 600:255:out
move moons enter turn    0:-0.12 1000:0:out
move moons wait  turn    0:0 4800:1
move medium enter turn   0:0.12 1000:0:out
move small  enter turn   0:0.12 1000:0:out
move tiny   enter turn   0:0.12 1000:0:out
move medium wait  turn   0:0 4800:-1
move small  wait  turn   0:0 4800:-1
move tiny   wait  turn   0:0 4800:-1
move eye   wait  tall    0:1 3000:1 3090:0.06:inout 3200:1:inout 4800:1
move veil  leave opacity 0:0 480:255:in
